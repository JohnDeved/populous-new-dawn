import { createRoot } from 'react-dom/client'
import { useEffect, useRef } from 'react'
import { hudScale } from '/tmp/populous-selector-scene-457/app/hud-layout.ts'
import BaselineResult from './BaselineResult'
import './baseline.css'
const status = new URLSearchParams(location.search).get('status') === 'lost' ? 'lost' : 'won'
function Fixture() {
  const shell = useRef<HTMLElement>(null)
  useEffect(() => {
    const resize = () => shell.current?.style.setProperty('--hud-scale', String(hudScale(innerWidth, innerHeight, 'auto')))
    resize()
    window.addEventListener('resize', resize)
    return () => window.removeEventListener('resize', resize)
  }, [])
  return <main className="game-shell" ref={shell}><div className="world-viewport" /><BaselineResult status={status} /></main>
}
createRoot(document.getElementById('root')!).render(<Fixture />)
